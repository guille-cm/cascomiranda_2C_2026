62static bool montar_sistema_fat_sd(void) {
63 if (s_sd_montada && s_card != NULL) {
64 return true;
65 }
66 if (s_intento_montaje_hecho) {
67 return false; // Evita re-registrar el dispositivo SPI si la tarjeta no está insertada
68 }
69 s_intento_montaje_hecho = true;
70
71 // 1. Configura GPIO_13 (CS) desconectando el transceptor USB_D+ interno
72 gpio_config_t cs_conf = {
73 .pin_bit_mask = (1ULL << MICROSD_DEFAULT_CS_GPIO),
74 .mode = GPIO_MODE_OUTPUT,
75 .pull_up_en = GPIO_PULLUP_ENABLE,
76 .pull_down_en = GPIO_PULLDOWN_DISABLE,
77 .intr_type = GPIO_INTR_DISABLE
78 };
79 gpio_config(&cs_conf);
80 gpio_set_level((gpio_num_t)MICROSD_DEFAULT_CS_GPIO, 1);
81
82 // 2. Pull-ups internos en MISO, MOSI y SCK
83 gpio_set_pull_mode((gpio_num_t)MICROSD_DEFAULT_MISO_GPIO, GPIO_PULLUP_ONLY);
84 gpio_set_pull_mode((gpio_num_t)MICROSD_DEFAULT_MOSI_GPIO, GPIO_PULLUP_ONLY);
85 gpio_set_pull_mode((gpio_num_t)MICROSD_DEFAULT_SCK_GPIO, GPIO_PULLUP_ONLY);
86
87 // 3. Inicializa el bus SPI2
88 if (!s_spi_bus_inicializado) {
89 spi_bus_config_t bus_cfg = {
90 .mosi_io_num = MICROSD_DEFAULT_MOSI_GPIO,
91 .miso_io_num = MICROSD_DEFAULT_MISO_GPIO,
92 .sclk_io_num = MICROSD_DEFAULT_SCK_GPIO,
93 .quadwp_io_num = -1,
94 .quadhd_io_num = -1,
95 .max_transfer_sz = 4096
96 };
97 if (spi_bus_initialize(SD_SPI_HOST_ID, &bus_cfg, SPI_DMA_CH_AUTO) != ESP_OK) {
98 return false;
99 }
100 s_spi_bus_inicializado = true;
101 }
102
103 // 4. Monta el volumen FAT16/FAT32 sobre SPI a 4 MHz
104 esp_vfs_fat_sdmmc_mount_config_t mount_config = {
105 .format_if_mount_failed = false,
106 .max_files = 3,
107 .allocation_unit_size = 16 * 1024
108 };
109
110 sdmmc_host_t host = SDSPI_HOST_DEFAULT();
111 host.slot = SD_SPI_HOST_ID;
112 host.max_freq_khz = SD_SPI_CLOCK_KHZ;
113
114 sdspi_device_config_t slot_config = SDSPI_DEVICE_CONFIG_DEFAULT();
115 slot_config.gpio_cs = (gpio_num_t)MICROSD_DEFAULT_CS_GPIO;
116 slot_config.host_id = SD_SPI_HOST_ID;
117
118 if (esp_vfs_fat_sdspi_mount(MICROSD_MOUNT_POINT, &host, &slot_config, &mount_config, &s_card) != ESP_OK) {
119 s_sd_montada = false;
120 s_card = NULL;
121 return false;
122 }
123
124 s_sd_montada = true;
125
126 // 5. Crea el encabezado de DATALOG.CSV si el archivo no existe o está vacío
127 struct stat st;
128 if (stat(MICROSD_CSV_FILEPATH, &st) != 0 || st.st_size == 0) {
129 FILE *f = fopen(MICROSD_CSV_FILEPATH, "w");
130 if (f != NULL) {
131 const char *hdr = "Fecha,Hora,FC,FR,Temp,Hum,Gases,Movimiento\n";
132 fwrite(hdr, 1, strlen(hdr), f);
133 fclose(f);
134 }
135 }
136
137 return true;
138}

141bool MicroSdInit(void) {
142 if (s_sd_mutex == NULL) {
143 s_sd_mutex = xSemaphoreCreateMutex();
144 }
145
146 // Inicializa el RTC interno en 2026-01-01 00:00:00 sin consumir el stack de app_main
147 struct timeval tv_now;
148 gettimeofday(&tv_now, NULL);
149 if (tv_now.tv_sec < EPOCH_BASE_DEFAULT) {
150 tv_now.tv_sec = EPOCH_BASE_DEFAULT;
151 tv_now.tv_usec = 0;
152 settimeofday(&tv_now, NULL);
153 }
154 return true;

157bool MicroSdEscribirRegistro(const registro_clinico_sd_t *reg) {
158 if (reg == NULL || s_sd_mutex == NULL) {
159 return false;
160 }
161
162 if (xSemaphoreTake(s_sd_mutex, pdMS_TO_TICKS(500)) != pdTRUE) {
163 return false;
164 }
165
166 if (!s_sd_montada) {
167 if (!montar_sistema_fat_sd()) {
168 xSemaphoreGive(s_sd_mutex);
169 return false;
170 }
171 }
172
173 char fecha[12];
174 char hora[10];
175 char linea_csv[80];
176 MicroSdObtenerFechaHoraActual(fecha, sizeof(fecha), hora, sizeof(hora));
177
178 // Fase 1: Apertura y escritura de la línea CSV
179 FILE *f = fopen(MICROSD_CSV_FILEPATH, "a");
180 if (f == NULL) {
181 esp_vfs_fat_sdcard_unmount(MICROSD_MOUNT_POINT, s_card);
182 s_card = NULL;
183 s_sd_montada = false;
184 s_intento_montaje_hecho = false;
185 xSemaphoreGive(s_sd_mutex);
186 return false;
187 }
188
189 int16_t t_ent = reg->temp_decimas_c / 10;
190 int16_t t_dec = abs(reg->temp_decimas_c % 10);
191 uint16_t h_ent = reg->hum_decimas_pct / 10;
192 uint16_t h_dec = reg->hum_decimas_pct % 10;
193
194 int len = snprintf(linea_csv, sizeof(linea_csv), "%s,%s,%u,%u,%d.%d,%u.%u,%u,%u\n",
195 fecha,
196 hora,
197 (unsigned int)reg->fc_bpm,
198 (unsigned int)reg->fr_rpm,
199 (int)t_ent, (int)t_dec,
200 (unsigned int)h_ent, (unsigned int)h_dec,
201 reg->alerta_gases ? 1U : 0U,
202 reg->movimiento_sala ? 1U : 0U);
203 if (len > 0) {
204 fwrite(linea_csv, 1, (size_t)len, f);
205 }
206
207 // Fase 2: Cierre del archivo y Flush inmediato a la tabla FAT16
208 fclose(f);
209
210 s_total_registros_escritos++;
211 xSemaphoreGive(s_sd_mutex);
212 return true;
213}

254int MicroSdLeerUltimosRegistrosJson(char *json_buf, size_t max_len, uint16_t max_registros) {
255 if (json_buf == NULL || max_len < 64) {
256 return 0;
257 }
258
259 if (s_sd_mutex == NULL || xSemaphoreTake(s_sd_mutex, pdMS_TO_TICKS(500)) != pdTRUE) {
260 return snprintf(json_buf, max_len, "{\"sd_ok\":false,\"rows\":[]}");
261 }
262
263 if (!s_sd_montada && !montar_sistema_fat_sd()) {
264 xSemaphoreGive(s_sd_mutex);
265 return snprintf(json_buf, max_len, "{\"sd_ok\":false,\"rows\":[]}");
266 }
267
268 // Fase 3: Apertura y lectura SPI desde el final del archivo con fseek
269 FILE *f = fopen(MICROSD_CSV_FILEPATH, "r");
270 if (f == NULL) {
271 xSemaphoreGive(s_sd_mutex);
272 return snprintf(json_buf, max_len, "{\"sd_ok\":false,\"rows\":[]}");
273 }
274
275 fseek(f, 0, SEEK_END);
276 long file_size = ftell(f);
277 long read_offset = 0;
278 if (file_size > SD_TAIL_READ_BYTES) {
279 read_offset = file_size - SD_TAIL_READ_BYTES;
280 }
281 fseek(f, read_offset, SEEK_SET);
282
283 size_t bytes_leidos = fread(s_tail_buf, 1, SD_TAIL_READ_BYTES, f);
284 s_tail_buf[bytes_leidos] = '\0';
285 fclose(f);
286
287 char *lineas[60];
288 uint16_t total_lineas = 0;
289 char *saveptr = NULL;
290 char *linea = strtok_r(s_tail_buf, "\r\n", &saveptr);
291
292 if (read_offset > 0 && linea != NULL) {
293 linea = strtok_r(NULL, "\r\n", &saveptr);
294 }
295
296 while (linea != NULL) {
297 if (strncmp(linea, "Fecha", 5) != 0 && strlen(linea) > 15) {
298 if (total_lineas < 60) {
299 lineas[total_lineas++] = linea;
300 } else {
301 memmove(&lineas[0], &lineas[1], 59 * sizeof(char *));
302 lineas[59] = linea;
303 }
304 }
305 linea = strtok_r(NULL, "\r\n", &saveptr);
306 }
307
308 uint16_t inicio = (total_lineas > max_registros) ? (total_lineas - max_registros) : 0;
309 size_t pos = snprintf(json_buf, max_len, "{\"sd_ok\":true,\"total\":%lu,\"rows\":[",
310 (unsigned long)s_total_registros_escritos);
311
312 bool primero = true;
313 for (uint16_t i = inicio; i < total_lineas; i++) {
314 char *p_save = NULL;
315 char *c_fecha = strtok_r(lineas[i], ",", &p_save);
316 char *c_hora = strtok_r(NULL, ",", &p_save);
317 char *c_fc = strtok_r(NULL, ",", &p_save);
318 char *c_fr = strtok_r(NULL, ",", &p_save);
319 char *c_temp = strtok_r(NULL, ",", &p_save);
320 char *c_hum = strtok_r(NULL, ",", &p_save);
321 char *c_gas = strtok_r(NULL, ",", &p_save);
322 char *c_mov = strtok_r(NULL, ",", &p_save);
323
324 if (c_fecha && c_hora && c_fc && c_fr && c_temp && c_hum && c_gas && c_mov) {
325 int escritos = snprintf(json_buf + pos, max_len - pos,
326 "%s[\"%s\",\"%s\",%s,%s,%s,%s,%s,%s]",
327 primero ? "" : ",",
328 c_fecha, c_hora, c_fc, c_fr, c_temp, c_hum, c_gas, c_mov);
329 if (escritos > 0 && (pos + (size_t)escritos + 4) < max_len) {
330 pos += (size_t)escritos;
331 primero = false;
332 } else {
333 break;
334 }
335 }
336 }
337
338 pos += snprintf(json_buf + pos, max_len - pos, "]}");
339 xSemaphoreGive(s_sd_mutex);
340 return (int)pos;
341}

343esp_err_t MicroSdEnviarCsvCompletoHttp(httpd_req_t *req) {
344 if (s_sd_mutex == NULL || xSemaphoreTake(s_sd_mutex, pdMS_TO_TICKS(1000)) != pdTRUE) {
345 httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "Bus SPI ocupado");
346 return ESP_FAIL;
347 }
348
349 FILE *f = fopen(MICROSD_CSV_FILEPATH, "r");
350 if (f == NULL) {
351 xSemaphoreGive(s_sd_mutex);
352 httpd_resp_send_err(req, HTTPD_404_NOT_FOUND, "DATALOG.CSV no encontrado");
353 return ESP_FAIL;
354 }
355
356 httpd_resp_set_type(req, "text/csv; charset=utf-8");
357 httpd_resp_set_hdr(req, "Content-Disposition", "attachment; filename=\"DATALOG.CSV\"");
358
359 size_t leidos = 0;
360 while ((leidos = fread(s_csv_chunk, 1, sizeof(s_csv_chunk), f)) > 0) {
361 if (httpd_resp_send_chunk(req, s_csv_chunk, leidos) != ESP_OK) {
362 break;
363 }
364 }
365 fclose(f);
366 xSemaphoreGive(s_sd_mutex);
367 httpd_resp_send_chunk(req, NULL, 0);
368 return ESP_OK;
369}
