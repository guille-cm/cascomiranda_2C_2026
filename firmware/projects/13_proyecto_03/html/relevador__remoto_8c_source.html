26void RelevadorRemotoInit(gpio_t pin) {
27 s_rele_pin = pin;
28
29 // Configuración del registro GPIO del ESP32-C6:
30 // - pin_bit_mask: Establece la máscara con el bit 8 en nivel alto (1ULL << GPIO_8).
31 // - mode: Modo entrada de alta impedancia (GPIO_MODE_INPUT).
32 // - pull_up_en: Habilita la resistencia pull-up interna complementaria (~45 kΩ) a 3.3V.
33 // - pull_down_en: Deshabilita la red de pull-down interna.
34 // - intr_type: Desactiva interrupciones para realizar muestreo determinístico por software.
35 gpio_config_t io_conf = {
36 .pin_bit_mask = (1ULL << s_rele_pin),
37 .mode = GPIO_MODE_INPUT,
38 .pull_up_en = GPIO_PULLUP_ENABLE,
39 .pull_down_en = GPIO_PULLDOWN_DISABLE,
40 .intr_type = GPIO_INTR_DISABLE
41 };
42 gpio_config(&io_conf);
43
44 // Muestreo inicial del estado del pin para inicializar la memoria de flanco
45 // Si el contacto está abierto, GPIORead devuelve true (1 lógico = 3.3V)
46 s_rele_estado_previo = GPIORead(s_rele_pin);
47}

49bool RelevadorRemotoEventoActivacion(void) {
50 // Lectura del registro de entrada digital GPIO:
51 // Retorna true (1) si el pin está en 3.3V (reposo), o false (0) si fue llevado a masa (0V).
52 bool lectura = GPIORead(s_rele_pin);
53
54 // Detección de flanco descendente (High -> Low):
55 // Se cumple cuando el estado instantáneo es 0V (!lectura es true) y en el ciclo anterior
56 // se encontraba en nivel alto de 3.3V (s_rele_estado_previo == true).
57 if (!lectura && s_rele_estado_previo) {
58 s_rele_estado_previo = false; // Memoriza el estado presionado para no re-disparar en bucle
59 return true; // Señaliza el evento a la tarea de FreeRTOS
60 } else if (lectura && !s_rele_estado_previo) {
61 // Flanco ascendente (Low -> High): El contacto se abrió y retornó a 3.3V
62 s_rele_estado_previo = true; // Restaura la memoria de flanco para permitir una nueva pulsación
63 }
64 return false; // Sin evento de transición activa
65}
