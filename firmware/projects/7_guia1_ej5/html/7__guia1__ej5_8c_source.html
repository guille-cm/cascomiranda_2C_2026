58 // Inicialización del hardware GPIO previo a su uso

61 }
62
63 uint8_t digito_prueba = 0;
64
65 while (1) {

69 bcdToGpio(digito_prueba, gpio_bcd_bus);
70
71 // Verificación por software (impresión teórica de los bits)
72 for (uint8_t i = 0; i < 4; i++){
73 // Usamos un desplazamiento de bits para imprimir cómo debería estar cada bit
74 uint8_t estado_bit = (digito_prueba & (1 << i)) ? 1 : 0;
75 printf("Bus b%d (Pin %d): %d\n", i, gpio_bcd_bus[i].pin, estado_bit);

77 printf("\n");
