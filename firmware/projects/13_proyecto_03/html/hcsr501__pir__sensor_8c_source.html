28void HcSr501PirInit(gpio_t pin) {
29 s_pir_pin = pin;
30 s_contador_detecciones = 0;
31
32 // Configuración del puerto de entrada para GPIO_12:
33 // - pin_bit_mask: Asigna el bit 12 en el registro de configuración (1ULL << GPIO_12).
34 // - mode: Entrada digital de alta impedancia (GPIO_MODE_INPUT).
35 // - pull_up_en: Deshabilitado (la salida del BISS0001 es activa, tipo push-pull).
36 // - pull_down_en: Habilitado para asegurar 0V cuando el módulo se desconecta.
37 // - intr_type: Desactivado para realizar muestreo sincrónico en FreeRTOS.
38 gpio_config_t io_conf = {
39 .pin_bit_mask = (1ULL << s_pir_pin),
40 .mode = GPIO_MODE_INPUT,
41 .pull_up_en = GPIO_PULLUP_DISABLE,
42 .pull_down_en = GPIO_PULLDOWN_ENABLE,
43 .intr_type = GPIO_INTR_DISABLE
44 };
45 gpio_config(&io_conf);
46
47 // Marca temporal del instante de inicialización (en microsegundos)
48 s_init_timestamp_us = esp_timer_get_time();
49
50 // Sincronización del estado previo con el nivel real del pin al encender
51 // Evita interpretar el pulso de 1.15 s del encendido como una detección de movimiento
52 s_estado_previo = GPIORead(s_pir_pin);
53 s_estado_actual = false;
54}

56bool HcSr501PirEventoMovimiento(void) {
57 // Lectura del bit de estado lógico en el pin GPIO_12
58 bool lectura_pin = GPIORead(s_pir_pin);
59
60 // Evaluación de la ventana de estabilización térmica y eléctrica (30 segundos):
61 // Si aún no transcurrieron 60.000.000 us, se actualiza el estado previo pero se retorna false
62 if ((esp_timer_get_time() - s_init_timestamp_us) < (int64_t)HCSR501_STARTUP_SETTLE_US) {
63 s_estado_previo = lectura_pin;
64 s_estado_actual = false;
65 return false;
66 }
67
68 s_estado_actual = lectura_pin;
69
70 // Detección de flanco ascendente (Low -> High):
71 // El pin pasa de 0V (reposo) a 3.3V (detección de calor y movimiento de personas)
72 if (s_estado_actual && !s_estado_previo) {
73 s_estado_previo = true; // Bloquea re-disparos mientras el pulso continúe en alto
74 s_contador_detecciones++; // Incrementa el acumulador total de eventos
75 return true; // Notifica el flanco ascendente al planificador
76 } else if (!s_estado_actual && s_estado_previo) {
77 // Flanco descendente (High -> Low): Finalizó el pulso de 1 segundo calibrado en el sensor
78 s_estado_previo = false; // Rehabilita la detección para el siguiente ingreso
79 }
80 return false;
81}
