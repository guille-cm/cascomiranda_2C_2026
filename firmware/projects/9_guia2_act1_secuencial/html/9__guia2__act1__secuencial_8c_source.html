106 bool medir_activado = true;
107 bool hold_activado = false;
108 uint16_t distancia = 0;
109 uint16_t distancia_display = 0;
110
111 // Variables para detección de flancos de las teclas
112 bool tec1_anterior = false;
113 bool tec2_anterior = false;
114
115 // Contador de tiempo
116 uint16_t timer_medicion = 0;
117
118 while (true) {
119 // --- LECTURA DE TECLAS (Se ejecuta rápido, cada 50ms) ---
120 uint8_t teclas = SwitchesRead();
121 bool tec1_actual = (teclas == SWITCH_1 || teclas == (SWITCH_1 | SWITCH_2));
122 bool tec2_actual = (teclas == SWITCH_2 || teclas == (SWITCH_1 | SWITCH_2));
123
124 // Detección de flanco para TEC1
125 if (tec1_actual && !tec1_anterior) {
126 medir_activado = !medir_activado; // Invierte estado solo al apretar
127 }
128 tec1_anterior = tec1_actual;
129
130 // Detección de flanco para TEC2
131 if (tec2_actual && !tec2_anterior) {
132 hold_activado = !hold_activado;
133 if (hold_activado) distancia_display = distancia; // Congela el valor actual
134 }
135 tec2_anterior = tec2_actual;
136
137
138 // --- LÓGICA DE MEDICIÓN (Se ejecuta solo 1 vez por segundo) ---
139 if (timer_medicion >= 1000) {
140 timer_medicion = 0; // Reiniciamos el cronómetro
141
142 if (medir_activado) {
143 distancia = HcSr04ReadDistanceInCentimeters();
144
145 // Lógica de LEDs según consigna
146 if (distancia < 10) {
147 LedsOffAll();
148 } else if (distancia >= 10 && distancia < 20) {
149 LedOn(LED_1); LedOff(LED_2); LedOff(LED_3);
150 } else if (distancia >= 20 && distancia < 30) {
151 LedOn(LED_1); LedOn(LED_2); LedOff(LED_3);
152 } else {
153 LedOn(LED_1); LedOn(LED_2); LedOn(LED_3);
154 }
155
156 // Lógica de Display
157 if (hold_activado) {
158 LcdItsE0803Write(distancia_display);
159 } else {
160 LcdItsE0803Write(distancia);
161 }
162 } else {
163 LedsOffAll();
164 LcdItsE0803Off();
165 }
166 }
167
168 // Retardo base del bucle. El sistema reacciona a los botones cada 50ms.
169 vTaskDelay(50 / portTICK_PERIOD_MS);
170 timer_medicion += 50;
171 }
172}
