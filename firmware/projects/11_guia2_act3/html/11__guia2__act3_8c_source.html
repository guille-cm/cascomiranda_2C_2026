139void FuncUartRecepcion(void *param) {
140 uint8_t letra;
141 UartReadByte(UART_PC, &letra);
142
143 if (letra == 'O' || letra == 'o') {
144 medir_activado = !medir_activado;
145 } else if (letra == 'H' || letra == 'h') {

150static void MedirTask(void *pvParameter) {
151 while (true) {
152 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
153 if (medir_activado) {
154 distancia = HcSr04ReadDistanceInCentimeters();
155
156 /* Envío por UART aprovechando el refresco de 1s (Timer A) de esta tarea.
157 Se formatea a: 3 dígitos ascii + espacio + "cm\r\n" */
158 if (!hold_activado) {
159 /* Si se requiere forzar SIEMPRE 3 caracteres (ej: "005", "045", "120"),
160 se puede usar sprintf, pero usaremos las funciones base de la cátedra */
161 UartSendString(UART_PC, (char*)UartItoa(distancia, 10));
162 UartSendString(UART_PC, " cm\r\n");
163 }
164 }
165 }
166}

168static void DisplayTask(void *pvParameter) {
169 while (true) {
170 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
171 if (medir_activado) {
172 /* Control de la barra de LEDs según la distancia */
173 if (distancia < 10) {
174 LedsOffAll();
175 } else if (distancia >= 10 && distancia < 20) {
176 LedOn(LED_1);
177 LedOff(LED_2);
178 LedOff(LED_3);
179 } else if (distancia >= 20 && distancia < 30) {
180 LedOn(LED_1);
181 LedOn(LED_2);
182 LedOff(LED_3);
183 } else if (distancia >= 30) {
184 LedOn(LED_1);
185 LedOn(LED_2);
186 LedOn(LED_3);
187 }
188
189 /* Control del LCD según la función HOLD */
190 if (!hold_activado) {
191 LcdItsE0803Write(distancia);
192 }
193 } else {
194 LedsOffAll();
195 LcdItsE0803Off();
196 }
197 }

201void app_main(void) {
202 /* Inicialización de periféricos básicos */
203 LedsInit();

213 .param_p = NULL
214 };
215 UartInit(&mi_uart);
216
217 /* Configuración de interrupciones de teclas físicas */
218 SwitchActivInt(SWITCH_1, FuncTecla1, NULL);
219 SwitchActivInt(SWITCH_2, FuncTecla2, NULL);
220
221 /* Configuración e inicialización de Timers */
222 timer_config_t timer_medicion = {
223 .timer = TIMER_A,
224 .period = CONFIG_BLINK_PERIOD_MEDICION,
225 .func_p = FuncTimerA,
226 .param_p = NULL
227 };
228 TimerInit(&timer_medicion);
229
230 timer_config_t timer_display = {
231 .timer = TIMER_B,
232 .period = CONFIG_BLINK_PERIOD_DISPLAY,
233 .func_p = FuncTimerB,
234 .param_p = NULL
235 };
236 TimerInit(&timer_display);
237
238 /* Creación de tareas FreeRTOS */
239 xTaskCreate(&MedirTask, "MedirTask", 2048, NULL, 5, &medir_task_handle);
240 xTaskCreate(&DisplayTask, "DisplayTask", 2048, NULL, 5, &display_task_handle);
241
242 /* Inicio de los temporizadores */
243 TimerStart(timer_medicion.timer);
244 TimerStart(timer_display.timer);
245}
