128static void DisplayDrawStep(uint8_t anod_pin, uint8_t digit, bool upper_half) {
129 // 1. Apagamos todos los ánodos primero para evitar "fantomeo"
130 GPIOOff(ANODO_1);
131 GPIOOff(ANODO_2);
132 GPIOOff(ANODO_3);
133 GPIOOff(ANODO_4);
134
135 // 2. Apagamos todos los cátodos (estado HIGH = LED apagado)
136 GPIOOn(CATODO_1);
137 GPIOOn(CATODO_2);
138 GPIOOn(CATODO_3);
139 GPIOOn(CATODO_4);
140
141 // 3. Obtenemos el patrón correspondiente
142 const uint8_t* patron;
143 if (upper_half) {
144 patron = upper_segments[digit];
145 } else {
146 patron = lower_segments[digit];
147 }
148
149 // 4. Encendemos los segmentos necesarios (estado LOW = LED encendido)
150 if (patron[0] == 1) GPIOOff(CATODO_1); // Segmento A o D
151 if (patron[1] == 1) GPIOOff(CATODO_2); // Segmento B o E
152 if (patron[2] == 1) GPIOOff(CATODO_3); // Segmento C o F
153 if (patron[3] == 1) GPIOOff(CATODO_4); // Segmento G
154
155 // 5. Encendemos el ánodo correspondiente al paso actual
156 GPIOOn(anod_pin);
157
158 // 6. Mantenemos encendido el paso para engañar al cerebro
159 // Un tiempo de 3 a 5 milisegundos dará un total de 12 a 20 ms, resultando en 50 a 83 Hz.
160 // vTaskDelay(pdMS_TO_TICKS(DISPLAY_REFRESH_MS));
161 vTaskDelay(pdMS_TO_TICKS(display_refresh_ms));
162}

166uint8_t Display7sx2Show(uint8_t number) {
167 // Verificamos que el número sea válido para dos dígitos
168 if (number > 99) {
169 return false;
170 }
171
172 // Separamos el número en decenas y unidades
173 uint8_t decena = number / 10;
174 uint8_t unidad = number % 10;
175
176 // Dibujamos secuencialmente cada una de las 4 partes del display matricial
177 // 1. Mitad superior de la unidad
178 DisplayDrawStep(ANODO_1, unidad, true);
179
180 // 2. Mitad inferior de la unidad
181 DisplayDrawStep(ANODO_2, unidad, false);
182
183 // 3. Mitad superior de la decena
184 DisplayDrawStep(ANODO_3, decena, true);
185
186 // 4. Mitad inferior de la decena
187 DisplayDrawStep(ANODO_4, decena, false);
188
189 return true;
190}
