409static esp_err_t get_status_handler(httpd_req_t *req) {
410 char fila0[WEB_LCD_TEXT_MAX_LEN + 1];
411 estado_unidad_paciente_t copia;
412
413 // 1. Obtiene el texto vigente de la Fila 0 del display LCD (BIENVENIDO!, Mensaje, Paciente o IP)
414 WebUnidadPacienteObtenerTextoFila0(fila0);
415
416 /*
417 * 2. Lectura atómica mediante sección crítica (portENTER_CRITICAL / portEXIT_CRITICAL):
418 * Deshabilita temporalmente las interrupciones y el cambio de contexto mientras se copia
419 * la estructura completa s_datos_actuales hacia la variable local 'copia', garantizando que
420 * TaskDisplayLCD no modifique ningún campo a mitad de la lectura.
421 */
422 portENTER_CRITICAL(&s_web_spinlock);
423 copia = s_datos_actuales;
424 portEXIT_CRITICAL(&s_web_spinlock);
425
426 /*
427 * 3. Desglose aritmético de punto fijo (décimas) a parte entera y dígito decimal:
428 * - t_ent = temp_decimas_c / 10 (ej. 254 -> 25), t_dec = abs(temp_decimas_c % 10) (ej. 254 -> 4).
429 * - h_ent = hum_decimas_pct / 10 (ej. 528 -> 52), h_dec = hum_decimas_pct % 10 (ej. 528 -> 8).
430 */
431 int16_t t_ent = copia.temp_decimas_c / 10;
432 int16_t t_dec = abs(copia.temp_decimas_c % 10);
433 uint16_t h_ent = copia.hum_decimas_pct / 10;
434 uint16_t h_dec = copia.hum_decimas_pct % 10;
435
436 // 4. Construcción del documento JSON sobre el buffer estático s_json_status_buf (en sección .bss)
437 snprintf(s_json_status_buf, sizeof(s_json_status_buf),
438 "{\"fc\":%u,\"fr\":%u,\"al_fc\":%s,\"al_fr\":%s,\"sim\":%u,"
439 "\"pat_req\":%s,"
440 "\"temp\":\"%d.%d\",\"hum\":\"%u.%u\","
441 "\"mq3_rdy\":%s,\"mq3_al\":%s,\"mq3_s\":%lu,"
442 "\"pir\":%s,\"pir_n\":%lu,"
443 "\"fifo_n\":%u,\"fifo_txt\":\"%s\","
444 "\"fila0\":\"%s\",\"ip\":\"%s\","
445 "\"sd_ok\":%s,\"sd_n\":%lu,\"rtc_ok\":%s,"
446 "\"fecha\":\"%s\",\"hora\":\"%s\"}",
447 copia.fc_bpm,
448 copia.fr_rpm,
449 copia.alerta_fc ? "true" : "false",
450 copia.alerta_fr ? "true" : "false",
451 copia.modo_sim_local,
452 copia.paciente_requiere_atencion ? "true" : "false",
453 t_ent, t_dec,
454 h_ent, h_dec,
455 copia.mq3_precalentado ? "true" : "false",
456 copia.mq3_alcohol_detectado ? "true" : "false",
457 (unsigned long)copia.mq3_segundos_bajo,
458 copia.pir_movimiento ? "true" : "false",
459 (unsigned long)copia.pir_contador_eventos,
460 copia.alertas_fifo_cant,
461 copia.alerta_fifo_actual,
462 fila0,
463 s_ip_asignada,
464 copia.sd_montada ? "true" : "false",
465 (unsigned long)copia.sd_registros_totales,
466 MicroSdRelojSincronizado() ? "true" : "false",
467 copia.fecha_actual,
468 copia.hora_actual);
469
470 // 5. Envía la respuesta JSON al navegador cliente
471 httpd_resp_set_type(req, "application/json; charset=utf-8");
472 httpd_resp_send(req, s_json_status_buf, HTTPD_RESP_USE_STRLEN);
473 return ESP_OK;
474}

671void WebUnidadPacienteInit(void) {
672 /*
673 * 1. Inicialización de la partición NVS (Non-Volatile Storage) en memoria Flash:
674 * Requerida por el driver Wi-Fi de ESP-IDF para almacenar datos de calibración de RF.
675 * Si la partición está llena o cambió de versión, se formatea e inicializa nuevamente.
676 */
677 esp_err_t ret = nvs_flash_init();
678 if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
679 nvs_flash_erase();
680 nvs_flash_init();
681 }
682
683 /*
684 * 2. Inicialización de la interfaz de red (esp_netif), del bucle de eventos por defecto
685 * del sistema y vinculación del adaptador Wi-Fi Station con la pila TCP/IP lwIP.
686 */
687 esp_netif_init();
688 esp_event_loop_create_default();
689 esp_netif_create_default_wifi_sta();
690
691 // 3. Reserva de recursos de hardware y tarea interna del driver Wi-Fi con configuración por defecto
692 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
693 esp_wifi_init(&cfg);
694
695 // 4. Registro de callbacks para escuchar eventos de capa de enlace (WIFI_EVENT) y de capa de red (IP_EVENT)
696 esp_event_handler_instance_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, NULL);
697 esp_event_handler_instance_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &wifi_event_handler, NULL, NULL);
698
699 /*
700 * 5. Carga de credenciales de red (SSID y Password) con umbral de seguridad mínimo WPA2-PSK,
701 * selección del modo Estación (WIFI_MODE_STA) y puesta en marcha del transceptor (esp_wifi_start).
702 */
703 wifi_config_t wifi_config = {
704 .sta = {
705 .ssid = WIFI_SSID_DEFAULT,
706 .password = WIFI_PASS_DEFAULT,
707 .threshold.authmode = WIFI_AUTH_WPA2_PSK,
708 },
709 };
710 esp_wifi_set_mode(WIFI_MODE_STA);
711 esp_wifi_set_config(WIFI_IF_STA, &wifi_config);
712 esp_wifi_start();
713
714 // 6. Puesta en marcha del servidor HTTP en el puerto 80
715 iniciar_servidor_http();
716}

734void WebUnidadPacienteObtenerTextoFila0(char *buffer_17_bytes) {
735 if (buffer_17_bytes == NULL) return;
736
737 /*
738 * Evaluación jerárquica bajo sección crítica:
739 * 1. Si aún no obtuvo dirección IPv4 del router (!s_conectado_wifi) -> "BIENVENIDO!".
740 * 2. Si el operador envió un mensaje clínico predefinido (s_mensaje_activo no vacío) -> muestra el mensaje.
741 * 3. Si el operador ingresó el nombre del paciente (s_nombre_paciente no vacío) -> muestra el nombre.
742 * 4. Por defecto (campos web vacíos y Wi-Fi conectado) -> muestra la dirección IPv4 asignada (s_ip_asignada).
743 */
744 portENTER_CRITICAL(&s_web_spinlock);
745 if (!s_conectado_wifi) {
746 strncpy(buffer_17_bytes, "BIENVENIDO!", WEB_LCD_TEXT_MAX_LEN);
747 } else if (s_mensaje_activo[0] != '\0') {
748 strncpy(buffer_17_bytes, s_mensaje_activo, WEB_LCD_TEXT_MAX_LEN);
749 } else if (s_nombre_paciente[0] != '\0') {
750 strncpy(buffer_17_bytes, s_nombre_paciente, WEB_LCD_TEXT_MAX_LEN);
751 } else {
752 strncpy(buffer_17_bytes, s_ip_asignada, WEB_LCD_TEXT_MAX_LEN);
753 }
754 buffer_17_bytes[WEB_LCD_TEXT_MAX_LEN] = '\0';
755 portEXIT_CRITICAL(&s_web_spinlock);
756}
