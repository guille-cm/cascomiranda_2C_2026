26void Mq3AlcInit(gpio_t pin) {
27 s_mq3_pin = pin;
28
29 // Registra el instante de arranque en microsegundos para la temporización de 60 s
30 s_init_timestamp_us = esp_timer_get_time();
31 s_en_estado_bajo = false;
32 s_low_start_timestamp_us = 0;
33
34 // Configuración del pin de entrada digital GPIO_19:
35 // El divisor resistivo fija 3.2V en reposo. Se activa el pull-up interno como protección
36 // para evitar lecturas flotantes en caso de desconexión del cable de señal.
37 GPIOInit(s_mq3_pin, GPIO_INPUT);
38 gpio_set_pull_mode((gpio_num_t)s_mq3_pin, GPIO_PULLUP_ONLY);
39}

41bool Mq3AlcPrecalentamientoListo(void) {
42 // Calcula la diferencia temporal respecto al arranque:
43 // 60.000.000 microsegundos = 60 segundos
44 int64_t transcurrido_us = esp_timer_get_time() - s_init_timestamp_us;

48bool Mq3AlcActualizarEstado(void) {
49 // 1. Verificación de seguridad térmica: Si está en el primer minuto, ignora cualquier señal
50 if (!Mq3AlcPrecalentamientoListo()) {
51 s_en_estado_bajo = false;
52 return false;
53 }
54
55 // 2. Muestreo lógico del pin GPIO_19:
56 // La salida del comparador LM393 conmuta a 0V (nivel bajo) al superar el umbral de gas.
57 // Invertir la lectura (!GPIORead) hace que dout_bajo sea true cuando hay presencia de gas.
58 bool dout_bajo = !GPIORead(s_mq3_pin);
59
60 if (dout_bajo) {
61 if (!s_en_estado_bajo) {
62 // Flanco de bajada físico: Inicia un nuevo evento de gas
63 s_en_estado_bajo = true;
64 s_low_start_timestamp_us = esp_timer_get_time(); // Guarda la marca de inicio
65 }
66 return true;
67 } else {
68 // Retorno a nivel alto (aire limpio): Restablece el cronómetro de detección
69 s_en_estado_bajo = false;
70 s_low_start_timestamp_us = 0;
71 return false;
72 }
73}

75uint32_t Mq3AlcObtenerDuracionPulsoSeg(void) {
76 if (!s_en_estado_bajo) {
77 return 0; // En reposo no hay duración acumulada
78 }
79
80 // Calcula el tiempo transcurrido desde el flanco de bajada y convierte de microsegundos a segundos
81 int64_t duracion_us = esp_timer_get_time() - s_low_start_timestamp_us;
82 if (duracion_us < 0) {
83 return 0;
84 }
85 return (uint32_t)(duracion_us / 1000000ULL);
86}
