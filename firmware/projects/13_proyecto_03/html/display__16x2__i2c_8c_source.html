105static void pcf8574_write(uint8_t data) {
106 /*
107 * Invoca la primitiva de escritura I2C pasando la dirección de 7 bits LCD_I2C_ADDR (0x27,
108 * que en el bus físico se transmite como (0x27 << 1) | 0 = 0x4E).
109 * Como el PCF8574 no posee registros internos direccionables, se duplica el argumento 'data'
110 * tanto en el campo de dirección de registro como en el de dato útil.
111 * El PCF8574 vuelca el primer byte 'data' a sus pines P7..P0 en el primer ACK y ratifica
112 * exactamente el mismo valor binario en el segundo ACK. A una frecuencia de reloj SCL de
113 * 100 kHz, esta transacción completa de 27 bits insume ~270 us, proporcionando de forma
114 * intrínseca el tiempo de establecimiento requerido por el controlador HD44780 sin bloquear
115 * temporizadores de hardware.
116 */
117 I2C_writeByte(LCD_I2C_ADDR, data, data);
118}

129static void send_nibble(uint8_t nibble, bool is_data) {
130 /*
131 * 1. Enmascaramiento y alineación del cuarteto de datos hacia el nibble superior (bits 7..4):
132 * - La operación (nibble & 0x0F) limpia por seguridad los 4 bits superiores (bits 7..4 = 0000),
133 * reteniendo únicamente los 4 bits inferiores útiles (bits 3..0).
134 * - El desplazamiento de 4 posiciones hacia la izquierda (<< 4) traslada esos 4 bits a las
135 * posiciones 7..4 del byte 'data', las cuales corresponden físicamente a las salidas P7..P4
136 * del PCF8574 (conectadas a los pines D7..D4 del módulo display 1602A Ver5.5).
137 * - Los 4 bits inferiores de 'data' (bits 3..0, salidas P3..P0) quedan inicialmente en 0000
138 * (lo que garantiza implícitamente que el bit P1 = MASK_RW permanezca en 0 = Modo Escritura).
139 */
140 uint8_t data = (uint8_t)((nibble & 0x0F) << 4);
141
142 /*
143 * 2. Configuración del bit P0 (Selección de Registro RS):
144 * - Si 'is_data' es true (envío de un carácter ASCII a la memoria gráfica DDRAM), se aplica
145 * una operación OR bit a bit con MASK_RS (0x01 = 00000001b), poniendo P0 en nivel ALTO (1).
146 * - Si 'is_data' es false (envío de una instrucción de control), el bit P0 permanece en BAJO (0).
147 */
148 if (is_data) {
149 data |= MASK_RS; // Activa el bit 0 (P0 -> RS = 1) para seleccionar el Registro de Datos
150 }
151
152 /*
153 * 3. Retención del bit P3 (Control de Retroiluminación BL):
154 * - Si la variable de estado 'backlight_state' es true, se aplica una operación OR bit a bit
155 * con MASK_BL (0x08 = 00001000b) para mantener polarizado el transistor que enciende el LED
156 * de luz de fondo del display durante toda la transferencia.
157 */
158 if (backlight_state) {
159 data |= MASK_BL; // Activa el bit 3 (P3 -> BL = 1) manteniendo encendida la retroiluminación
160 }
161
162 /*
163 * 4. Primera transferencia I2C (Establecimiento de datos y flanco ascendente de Enable):
164 * - La operación OR bit a bit (data | MASK_EN) pone en 1 el bit 2 (0x04 = 00000100b -> P2 = 1).
165 * - Al transmitirse por I2C, el PCF8574 presenta simultáneamente el cuarteto de datos en D7..D4,
166 * el nivel de RS en el pin 4 y eleva el pin 6 (Enable) a +5V durante ~270 us.
167 */
168 pcf8574_write(data | MASK_EN);
169
170 /*
171 * 5. Segunda transferencia I2C (Flanco descendente de Enable y captura en el HD44780):
172 * - El operador de complemento a uno (~MASK_EN) invierte los bits de 0x04 (00000100b),
173 * produciendo la máscara 0xFB (11111011b).
174 * - La operación AND bit a bit (data & ~MASK_EN) fuerza a 0 únicamente el bit 2 (P2 -> EN = 0),
175 * conservando intactos los bits de datos (P7..P4), el bit de Backlight (P3) y el bit RS (P0).
176 * - Esta transición de 1 a 0 en el pin Enable genera el flanco descendente que ordena al
177 * controlador HD44780 latchear internamente el cuarteto presente en D7..D4.
178 */
179 pcf8574_write(data & (uint8_t)(~MASK_EN));
180}

190static void send_byte(uint8_t value, bool is_data) {
191 /*
192 * En modo de interfaz de 4 bits, el protocolo del controlador HD44780 exige recibir primero
193 * los 4 bits más significativos (nibble alto, bits 7..4) y luego los 4 bits menos significativos
194 * (nibble bajo, bits 3..0):
195 * 1. (value >> 4): Desplaza el byte 4 posiciones hacia la derecha, ubicando los bits 7..4
196 * en las posiciones 3..0 para ser procesados y enviados en primer término por send_nibble().
197 */
198 send_nibble(value >> 4, is_data);
199
200 /*
201 * 2. (value & 0x0F): Aplica una máscara AND con 00001111b para aislar los bits 3..0 originales
202 * del byte y transmitirlos en segundo término, completando la reconstrucción del byte de 8 bits
203 * dentro del registro interno del controlador HD44780.
204 */
205 send_nibble(value & 0x0F, is_data);
206}

215 send_byte(cmd, false);

246bool Display16x2I2cInit(void) {
247 // Configura el periférico I2C maestro del ESP32-C6 a una velocidad de reloj SCL de 100 kHz
248 I2C_initialize(I2C_CLOCK_RATE_HZ);
249 // Cede la CPU en estado Blocked durante 100 ms hasta que la fuente de +5V supere los 4.5V requeridos por el HD44780
250 vTaskDelay(pdMS_TO_TICKS(100));
251
252 // Paso 1 de sincronización (Reset por instrucción): Envía el nibble 0x03 (D7..D4 = 0011b, equivalente a Function Set 8 bits)
253 // Trama en bus I2C: [0x4E, 0x3C, 0x3C] (EN=1) seguido de [0x4E, 0x38, 0x38] (EN=0)
254 send_nibble(0x03, false);
255 // Espera de 10 ms en estado Blocked (la hoja de datos del HD44780 exige un mínimo de 4.1 ms tras el primer 0x03)
256 vTaskDelay(pdMS_TO_TICKS(10));
257
258 // Paso 2 de sincronización: Segundo envío del nibble 0x03 para asegurar alineación si el display estaba a mitad de un byte en 4 bits
259 send_nibble(0x03, false);
260 // Espera de 2 ms en estado Blocked (la hoja de datos exige un mínimo de 100 us tras el segundo 0x03)
261 vTaskDelay(pdMS_TO_TICKS(2));
262
263 // Paso 3 de sincronización: Tercer envío del nibble 0x03; a partir de aquí el HD44780 se encuentra garantizadamente en modo de 8 bits
264 send_nibble(0x03, false);
265 // Espera de 2 ms antes de conmutar el ancho de bus
266 vTaskDelay(pdMS_TO_TICKS(2));
267
268 // Conmutación definitiva a interfaz de 4 bits: Envía un único nibble 0x02 (D7..D4 = 0010b -> DL = 0)
269 // Trama en bus I2C: [0x4E, 0x2C, 0x2C] (EN=1) seguido de [0x4E, 0x28, 0x28] (EN=0)
270 send_nibble(0x02, false);
271 // Espera de 2 ms para que el controlador interno reconfigure su máquina de estados a recepción por pares de nibbles
272 vTaskDelay(pdMS_TO_TICKS(2));
273
274 // Envía ya en modo de 4 bits (2 nibbles por comando) la configuración de formato 0x28 (DL=0: 4 bits, N=1: 2 filas, F=0: 5x8 puntos)
275 send_command(CMD_FUNCTION_SET_4B);
276 // Envía el comando 0x0C (D=1: enciende cristal líquido, C=0: apaga subrayado de cursor, B=0: apaga parpadeo de celda)
277 send_command(CMD_DISPLAY_ON);
278 // Ejecuta el comando 0x01 para escribir el código ASCII 0x20 (espacio) en las 80 posiciones de la memoria DDRAM y esperar 5 ms
279 Display16x2I2cClear();
280 // Envía el comando 0x06 (I/D=1: incrementa automáticamente la dirección DDRAM tras cada carácter, S=0: pantalla estática)
281 send_command(CMD_ENTRY_MODE_SET);
282
283 return true;
284}

291void Display16x2I2cClear(void) {
292 // Transmite el comando 0x01 (nibble alto 0x0 y nibble bajo 0x1 con RS=0)

295 vTaskDelay(pdMS_TO_TICKS(5));

305void Display16x2I2cSetCursor(uint8_t row, uint8_t col) {
306 // Acota por seguridad el índice de columna al límite máximo físico de la pantalla (columna 15)
307 if (col >= LCD_COLS) {
308 col = LCD_COLS - 1;
309 }
310 /*
311 * Cálculo de la instrucción Set DDRAM Address (bit 7 = 1, bits 6..0 = dirección de celda):
312 * - En el controlador HD44780, las 16 celdas visibles de la Fila 0 mapean en las direcciones
313 * DDRAM 0x00 a 0x0F (con el bit 7 en 1: 0x80 + col -> rango 0x80 a 0x8F).
314 * - Las 16 celdas visibles de la Fila 1 mapean en las direcciones DDRAM 0x40 a 0x4F
315 * (con el bit 7 en 1: 0xC0 + col -> rango 0xC0 a 0xCF).
316 */
317 uint8_t address = (row == LCD_ROW_0) ? (CMD_SET_DDRAM_ROW0 + col) : (CMD_SET_DDRAM_ROW1 + col);
318 // Envía el byte de posicionamiento al Registro de Instrucciones (RS = 0)
319 send_command(address);
320}

338void Display16x2I2cWriteString(const char *str) {
339 // Validación contra puntero nulo para prevenir accesos inválidos a memoria
340 if (str == NULL) return;
341 // Recorre byte a byte la cadena hasta hallar el delimitador nulo ('\0' = 0x00)
342 while (*str != '\0') {
343 Display16x2I2cWriteChar(*str); // Envía el carácter apuntado actualmente (el HD44780 avanza el cursor solo)
344 str++; // Incrementa el puntero al siguiente carácter en memoria
345 }
346}

357void Display16x2I2cWriteLine(uint8_t row, const char *str) {
358 // 1. Envía el comando de posicionamiento al extremo izquierdo de la fila (columna 0) -> 1 byte = 4 tramas I2C
359 Display16x2I2cSetCursor(row, 0);
360 uint8_t i = 0;
361
362 // 2. Transmite los caracteres válidos de la cadena sin exceder las 16 columnas físicas de la fila
363 if (str != NULL) {
364 while (str[i] != '\0' && i < LCD_COLS) {
365 Display16x2I2cWriteChar(str[i]);
366 i++;
367 }
368 }
369
370 // 3. Si la cadena tenía menos de 16 caracteres, rellena las columnas restantes con espacios (' ' = 0x20)
371 // para sobrescribir y limpiar cualquier carácter perteneciente a un mensaje anterior más largo
372 while (i < LCD_COLS) {
373 Display16x2I2cWriteChar(' ');
374 i++;
375 }
376}

383void Display16x2I2cSetBacklight(bool state) {
384 // Actualiza la bandera estática para que las futuras llamadas a send_nibble() conserven este estado
385 backlight_state = state;
386 // Selecciona la máscara 0x08 (P3=1, resto de pines en 0) si se enciende, o 0x00 si se apaga
387 uint8_t data = backlight_state ? MASK_BL : 0x00;
388 // Aplica inmediatamente el cambio en las salidas físicas del expansor PCF8574
389 pcf8574_write(data);
390}

413void Display16x2I2cTestPulseTrain(void) {
414 // Tabla de máscaras de un único bit activo (walking one) para excitar individualmente cada salida P0..P7
415 const uint8_t pulse_sequence[] = {
416 0x01, /* Bit 0 (00000001b) -> Salida P0: Pin 4 del display en ALTO (RS) */
417 0x02, /* Bit 1 (00000010b) -> Salida P1: Pin 5 del display en ALTO (RW) */
418 0x04, /* Bit 2 (00000100b) -> Salida P2: Pin 6 del display en ALTO (EN) */
419 0x08, /* Bit 3 (00001000b) -> Salida P3: Pin 16 del display en BAJO (Backlight encendido) */
420 0x10, /* Bit 4 (00010000b) -> Salida P4: Pin 11 del display en ALTO (D4) */
421 0x20, /* Bit 5 (00100000b) -> Salida P5: Pin 12 del display en ALTO (D5) */
422 0x40, /* Bit 6 (01000000b) -> Salida P6: Pin 13 del display en ALTO (D6) */
423 0x80 /* Bit 7 (10000000b) -> Salida P7: Pin 14 del display en ALTO (D7) */
424 };
425
426 // Recorre las 8 posiciones enviando la trama I2C y manteniendo cada estado durante 500 ms
427 for (uint8_t i = 0; i < 8; i++) {
428 pcf8574_write(pulse_sequence[i]);
429 vTaskDelay(pdMS_TO_TICKS(500)); /* Retardo no bloqueante de 500 ms para observación en osciloscopio */
430 }
431}
