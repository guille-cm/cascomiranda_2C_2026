599static void FifoEncolarAlerta(tipo_alerta_id_t id) {
600 // Descarta inserciones del identificador nulo
601 if (id == ALERTA_ID_NINGUNA) return;
602
603 // Ingresa a sección crítica deshabilitando interrupciones para acceso atómico a la cola compartida
604 portENTER_CRITICAL(&s_fifo_spinlock);
605 bool ya_existe = false;
606 // Recorre las 's_fifo_count' posiciones ocupadas a partir del índice de cabeza (s_fifo_head)
607 for (uint8_t i = 0; i < s_fifo_count; i++) {
608 uint8_t idx = (s_fifo_head + i) % FIFO_MAX_ALERTAS;
609 if (s_cola_fifo[idx] == id) {
610 ya_existe = true;
611 break;
612 }
613 }
614 // Si la alerta no estaba duplicada y existe espacio libre en el buffer circular, la inserta en s_fifo_tail
615 if (!ya_existe && s_fifo_count < FIFO_MAX_ALERTAS) {
616 s_cola_fifo[s_fifo_tail] = id;
617 s_fifo_tail = (s_fifo_tail + 1) % FIFO_MAX_ALERTAS;
618 s_fifo_count++;
619 }
620 // Libera el spinlock y restaura el estado de las interrupciones del sistema
621 portEXIT_CRITICAL(&s_fifo_spinlock);
622}

630static void FifoDesencolarMasAntigua(void) {
631 portENTER_CRITICAL(&s_fifo_spinlock);
632 if (s_fifo_count > 0) {
633 // Avanza el puntero de cabeza al siguiente elemento más antiguo envolviendo en FIFO_MAX_ALERTAS (8)
634 s_fifo_head = (s_fifo_head + 1) % FIFO_MAX_ALERTAS;
635 s_fifo_count--;
636 }
637 portEXIT_CRITICAL(&s_fifo_spinlock);
638}

715static void TaskGenerarECG(void *pvParameter) {
716 // Acumuladores de fase en punto fijo (módulo 10.000) e índices de recorrido de tabla (0 a 230)
717 static uint32_t acumulador_fase_ecg = 0;
718 static uint16_t indice_ecg = 0;
719
720 static uint32_t acumulador_fase_resp = 0;
721 static uint16_t indice_resp = 0;
722
723 while (1) {
724 /*
725 * Bloqueo por notificación directa de tarea:
726 * La tarea pasa a estado Blocked (consumo 0% de CPU) con espera infinita (portMAX_DELAY)
727 * hasta que la ISR FuncTimerA incrementa su valor de notificación cada 2000 us.
728 * El argumento pdTRUE limpia el contador de notificación a 0 al despertar (comportamiento de semáforo binario).
729 */
730 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
731
732 // 1. Síntesis DDS y actualización del modulador Sigma-Delta de ECG en GPIO_0
733 acumulador_fase_ecg += (fc_actual * 77);
734 while (acumulador_fase_ecg >= 10000) {
735 acumulador_fase_ecg -= 10000;
736 indice_ecg++;
737 // Si alcanza el final de las 231 muestras, envuelve al inicio del ciclo cardíaco
738 if (indice_ecg >= BUFFER_SIZE) {
739 indice_ecg = 0;
740 }
741 // Al pasar exactamente por el índice del pico R (133), levanta la bandera para destellar el LED 1
742 if (indice_ecg == R_PEAK_INDEX) {
743 pulso_detectado = true;
744 }
745 }
746 // Envía la muestra actual (0 a 255) al modulador Sigma-Delta 0 (conectado al filtro RC en GPIO_0)
747 AnalogOutputWrite(ecg[indice_ecg]);
748
749 // 2. Síntesis DDS y actualización del modulador Sigma-Delta de Respirograma en GPIO_18
750 acumulador_fase_resp += (fr_actual * 77);
751 while (acumulador_fase_resp >= 10000) {
752 acumulador_fase_resp -= 10000;
753 indice_resp++;
754 // Si alcanza el final de las 231 muestras, envuelve al inicio del ciclo ventilatorio
755 if (indice_resp >= BUFFER_SIZE) {
756 indice_resp = 0;
757 }
758 // Al pasar por el pico de máxima inspiración (92), levanta la bandera para destellar el LED 2
759 if (indice_resp == RESP_PEAK_INDEX) {
760 pulso_resp_detectado = true;
761 }
762 }
763 // Envía la muestra actual (40 a 235) al modulador Sigma-Delta 1 (conectado al filtro RC en GPIO_18)
764 AnalogOutput2Write(respirograma[indice_resp]);
765 }
766}

782static void TaskLeerECG_UART(void *pvParameter) {
783 uint32_t valor_adc_ecg = 0;
784 uint32_t valor_adc_resp = 0;
785 uint32_t valor_pot = 0;
786 uint8_t divisor_ciclos = 0;
787 char trama_uart[40];
788
789 while (1) {
790 // Permanece en estado Blocked hasta recibir la notificación directa desde la ISR FuncTimerB (cada 2 ms)
791 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
792
793 // 1. Conversión analógico-digital de la señal ECG filtrada por hardware en CH1 (GPIO_1)
794 AnalogInputReadSingle(CH1, (uint16_t*)&valor_adc_ecg);
795
796 divisor_ciclos++;
797
798 // 2. Cada 10 interrupciones (20 ms = 50 Hz) digitaliza CH3 (GPIO_3) y unifica ECG + Resp en una sola escritura UART
799 if ((divisor_ciclos % 10) == 0) {
800 AnalogInputReadSingle(CH3, (uint16_t*)&valor_adc_resp);
801 snprintf(trama_uart, sizeof(trama_uart), "\r\n>ecg:%u\r\n>resp:%u\r\n",
802 (uint16_t)valor_adc_ecg, (uint16_t)valor_adc_resp);
803 UartSendString(UART_PC, trama_uart);
804 } else {
805 // En los 9 ciclos intermedios transmite únicamente la muestra de alta velocidad del ECG
806 snprintf(trama_uart, sizeof(trama_uart), "\r\n>ecg:%u\r\n", (uint16_t)valor_adc_ecg);
807 UartSendString(UART_PC, trama_uart);
808 }
809
810 // 3. Cada 25 interrupciones (50 ms = 20 Hz) digitaliza el potenciómetro en CH2 (GPIO_2) sin colisión de periférico ADC
811 if (divisor_ciclos >= 25) {
812 divisor_ciclos = 0;
813 AnalogInputReadSingle(CH2, (uint16_t*)&valor_pot);
814 pot_raw_compartido = (uint16_t)valor_pot;
815 }
816 }

837static void TaskPotenciometro_LED(void *pvParameter) {
838 uint8_t tiempo_destello_ecg_ms = 0;
839 uint8_t tiempo_destello_resp_ms = 0;
840 uint32_t pot_filtrado = 1650;
841 bool en_alerta_fc = false;
842 bool en_alerta_fr = false;
843
844 bool tec1_anterior = false;
845 bool tec2_anterior = false;
846
847 while (1) {
848 // 1. Lectura de SWITCH_1 (GPIO_4) y SWITCH_2 (GPIO_15) con detección de flanco y exclusión mutua
849 uint8_t teclas = SwitchesRead();
850 bool tec1_actual = (teclas == SWITCH_1 || teclas == (SWITCH_1 | SWITCH_2));
851 bool tec2_actual = (teclas == SWITCH_2 || teclas == (SWITCH_1 | SWITCH_2));
852
853 // Flanco ascendente en Tecla 1: sólo conmuta entre MODO_SIM_NINGUNO y MODO_SIM_FC (ignorado si está en MODO_SIM_FR)
854 if (tec1_actual && !tec1_anterior) {
855 if (modo_simulacion == MODO_SIM_NINGUNO) {
856 modo_simulacion = MODO_SIM_FC;
857 } else if (modo_simulacion == MODO_SIM_FC) {
858 modo_simulacion = MODO_SIM_NINGUNO;
859 }
860 }
861 tec1_anterior = tec1_actual;
862
863 // Flanco ascendente en Tecla 2: sólo conmuta entre MODO_SIM_NINGUNO y MODO_SIM_FR (ignorado si está en MODO_SIM_FC)
864 if (tec2_actual && !tec2_anterior) {
865 if (modo_simulacion == MODO_SIM_NINGUNO) {
866 modo_simulacion = MODO_SIM_FR;
867 } else if (modo_simulacion == MODO_SIM_FR) {
868 modo_simulacion = MODO_SIM_NINGUNO;
869 }
870 }
871 tec2_anterior = tec2_actual;
872
873 // 2. Gestión de GPIO_8 (relevador remoto 433 MHz) y botón Web para llamado de atención / desencolado FIFO
874 bool pulsador_gpio8 = RelevadorRemotoEventoActivacion();
875 bool req_clear_web = WebUnidadPacienteConsumirSolicitudClearAlert();
876
877 if (pulsador_gpio8) {
878 uint8_t total_fifo = 0;
879 FifoConsultarFrente(&total_fifo);
880 if (total_fifo == 0) {
881 // Cola vacía: el paciente solicita asistencia -> Encola alerta, fija LED 3 y lanza melodía DRAMA
882 s_llamado_paciente_activo = true;
883 FifoEncolarAlerta(ALERTA_ID_LLAMADO_PACIENTE);
884 AvisoSonoroReproducirRtttl(songDrama_version_guille_cm);
885 } else {
886 // Cola con alertas activas: desencola la más antigua, detiene la melodía en curso y actualiza estado
887 FifoDesencolarMasAntigua();
888 AvisoSonoroDetenerRtttl();
889 if (!FifoContieneAlerta(ALERTA_ID_LLAMADO_PACIENTE)) {
890 s_llamado_paciente_activo = false;
891 }
892 }
893 }
894
895 // Procesamiento del botón remoto "Desactivar Alerta más Antigua (FIFO)" presionado en el navegador web
896 if (req_clear_web) {
897 FifoDesencolarMasAntigua();
898 AvisoSonoroDetenerRtttl();
899 if (!FifoContieneAlerta(ALERTA_ID_LLAMADO_PACIENTE)) {
900 s_llamado_paciente_activo = false;
901 }
902 }
903
904 // 3. Solicitudes Web de aviso sonoro bajo demanda (Melodía RTTTL de advertencia o Voz LPCM "Ayuda en camino")
905 if (WebUnidadPacienteConsumirSolicitudAdvertencia3s()) {
906 AvisoSonoroReproducirRtttl(songPacman_version_guille_cm);
907 }
908 if (WebUnidadPacienteConsumirSolicitudVozAyuda()) {
909 AvisoSonoroReproducirVozAyuda();
910 }
911
912 /*
913 * 4. Filtro digital pasabajos IIR exponencial sobre la lectura compartida del potenciómetro:
914 * Ecuación en diferencias: y[k] = (7/8)*y[k-1] + (1/8)*x[k].
915 * Suaviza las transiciones mecánicas y elimina el jitter del bit menos significativo del ADC.
916 */
917 pot_filtrado = (pot_filtrado * 7 + pot_raw_compartido) / 8;
918
919 // 5. Actualización selectiva de FC (40 a 140 bpm) o FR (8 a 24 rpm) mediante interpolación lineal entera
920 if (modo_simulacion == MODO_SIM_FC) {
921 uint16_t fc_calculada = FC_MIN + (pot_filtrado * (FC_MAX - FC_MIN)) / ADC_MAX_VAL;
922 if (fc_calculada > FC_MAX) fc_calculada = FC_MAX;
923 if (fc_calculada < FC_MIN) fc_calculada = FC_MIN;
924 fc_actual = fc_calculada;
925 } else if (modo_simulacion == MODO_SIM_FR) {
926 uint16_t fr_calculada = FR_MIN + (pot_filtrado * (FR_MAX - FR_MIN)) / ADC_MAX_VAL;
927 if (fr_calculada > FR_MAX) fr_calculada = FR_MAX;
928 if (fr_calculada < FR_MIN) fr_calculada = FR_MIN;
929 fr_actual = fr_calculada;
930 }
931
932 // 6. Evaluación de rango fisiológico normal de FC (50 a 130 bpm) con banda de histéresis interna de 2 bpm
933 if (!en_alerta_fc) {
934 if (fc_actual < FC_NORMAL_MIN || fc_actual > FC_NORMAL_MAX) {
935 en_alerta_fc = true;
936 FifoEncolarAlerta(ALERTA_ID_ECG);
937 }
938 } else {
939 // Para salir del estado de alerta exige reingresar al intervalo [52, 128] bpm, evitando oscilaciones en el borde
940 if (fc_actual >= (FC_NORMAL_MIN + 2) && fc_actual <= (FC_NORMAL_MAX - 2)) {
941 en_alerta_fc = false;
942 }
943 }
944 alerta_fc_activa = en_alerta_fc;
945
946 // 7. Evaluación de rango fisiológico normal de FR (12 a 20 rpm) con banda de histéresis interna de 1 rpm
947 if (!en_alerta_fr) {
948 if (fr_actual < FR_NORMAL_MIN || fr_actual > FR_NORMAL_MAX) {
949 en_alerta_fr = true;
950 FifoEncolarAlerta(ALERTA_ID_RESPIRAT);
951 }
952 } else {
953 // Para salir del estado de alerta exige reingresar al intervalo [13, 19] rpm
954 if (fr_actual >= (FR_NORMAL_MIN + 1) && fr_actual <= (FR_NORMAL_MAX - 1)) {
955 en_alerta_fr = false;
956 }
957 }
958 alerta_fr_activa = en_alerta_fr;
959
960 // 8. Control de LED 1 (GPIO_11 - Frecuencia Cardíaca): Fijo en alerta, o destello de 60 ms (3 ciclos de 20 ms) por latido
961 if (en_alerta_fc) {
962 LedOn(LED_1);
963 pulso_detectado = false;
964 tiempo_destello_ecg_ms = 0;
965 } else {
966 if (pulso_detectado) {
967 pulso_detectado = false;
968 LedOn(LED_1);
