40/*==================[internal functions definition]==========================*/
45static void IRAM_ATTR dht22_gpio_negedge_isr(void *arg) {
46 if (!s_captura_activa) {
47 return;
48 }
49 // Lectura del temporizador de hardware de 64 bits en microsegundos (toma menos de 1 us)
50 int64_t now_us = esp_timer_get_time();
51 uint8_t idx = s_edge_count;
52 if (idx < DHT22_MAX_EDGES) {
53 s_edge_timestamps[idx] = now_us; // Guarda la marca de tiempo del flanco descendente
54 s_edge_count = idx + 1; // Avanza el puntero de flancos capturados
55 }
56}

59void Dht22Init(gpio_t pin) {
60 s_dht22_pin = pin;
61
62 // Configuración inicial del puerto GPIO_9 como entrada con pull-up
63 gpio_config_t io_conf = {
64 .pin_bit_mask = (1ULL << s_dht22_pin),
65 .mode = GPIO_MODE_INPUT,
66 .pull_up_en = GPIO_PULLUP_ENABLE,
67 .pull_down_en = GPIO_PULLDOWN_DISABLE,
68 .intr_type = GPIO_INTR_NEGEDGE // Disparo por flanco descendente (High -> Low)
69 };
70 gpio_config(&io_conf);
71
72 // Instalación única del servicio de interrupciones de GPIO en el sistema operativo
73 if (!s_isr_registrada) {
74 gpio_install_isr_service(0);
75 gpio_isr_handler_add((gpio_num_t)s_dht22_pin, dht22_gpio_negedge_isr, NULL);
76 s_isr_registrada = true;
77 }
78 // Mantiene deshabilitada la interrupción hasta que se inicie una transacción de lectura
79 gpio_intr_disable((gpio_num_t)s_dht22_pin);
80}

82bool Dht22Read(dht22_data_t *out_data) {
83 if (out_data == NULL) {
84 return false;
85 }
86
87 uint8_t raw_bytes[5] = {0, 0, 0, 0, 0};
88
89 // 1. Deshabilita la interrupción durante la generación del pulso de arranque
90 s_captura_activa = false;
91 gpio_intr_disable((gpio_num_t)s_dht22_pin);
92
93 // 2. Conmuta el pin a SALIDA y fija nivel BAJO (0V) durante 2000 us (2 ms)
94 // Esto le indica al microcontrolador interno del sensor que despierte y prepare la trama
95 gpio_set_direction((gpio_num_t)s_dht22_pin, GPIO_MODE_OUTPUT);
96 gpio_set_level((gpio_num_t)s_dht22_pin, 0);
97 esp_rom_delay_us(DHT22_START_LOW_US);
98
99 // 3. Eleva la línea a 3.3V durante 25 us para generar un flanco de subida nítido
100 gpio_set_level((gpio_num_t)s_dht22_pin, 1);
101 esp_rom_delay_us(25);
102
103 // 4. Conmuta inmediatamente a ENTRADA de alta impedancia con pull-up activo
104 // El sensor toma el control de la línea y responde con el pulso de presencia (ACK)
105 gpio_set_direction((gpio_num_t)s_dht22_pin, GPIO_MODE_INPUT);
106 gpio_set_pull_mode((gpio_num_t)s_dht22_pin, GPIO_PULLUP_ONLY);
107
108 // 5. Reinicia el contador de flancos y habilita la captura por ISR
109 s_edge_count = 0;
110 s_captura_activa = true;
111 gpio_intr_enable((gpio_num_t)s_dht22_pin);
112
113 // 6. Cede la CPU a otras tareas durante 20 ms mediante FreeRTOS
114 // La trama completa tarda ~4.25 ms en transmitirse por hardware
115 vTaskDelay(pdMS_TO_TICKS(20));
116
117 // 7. Cierra la ventana de captura por interrupción
118 gpio_intr_disable((gpio_num_t)s_dht22_pin);
119 s_captura_activa = false;
120
121 uint8_t total_edges = s_edge_count;
122 // Se requieren como mínimo 41 flancos descendentes: 1 de ACK + 40 de datos
123 if (total_edges < 41) {
124 out_data->lectura_valida = false;
125 return false;
126 }
127
128 // 8. Decodificación bit a bit midiendo el período entre flancos descendentes sucesivos:
129 // first_bit_edge apunta al primer flanco correspondiente a los datos
130 uint8_t first_bit_edge = total_edges - 40;
131 for (uint8_t i = 0; i < 40; i++) {
132 // t_prev: tiempo del flanco descendente que inició el bit actual
133 // t_curr: tiempo del flanco descendente que finalizó el bit actual
134 int64_t t_prev = s_edge_timestamps[first_bit_edge + i - 1];
135 int64_t t_curr = s_edge_timestamps[first_bit_edge + i];
136 int32_t period_us = (int32_t)(t_curr - t_prev);
137
138 // Desplaza 1 bit a la izquierda para hacer lugar al nuevo bit en el byte correspondiente
139 raw_bytes[i / 8] <<= 1;
140
141 // Si el período entre flancos superó los 100 us, el tiempo en alto fue de ~70 us -> Bit '1'
142 // Si fue menor a 100 us (típicamente ~76 us), el tiempo en alto fue de ~28 us -> Bit '0'
143 if (period_us > DHT22_PERIOD_THRESHOLD_US) {
144 raw_bytes[i / 8] |= 0x01; // Inserta un '1' en el bit menos significativo
145 }
146 }
147
148 // 9. Comprobación del byte de Checksum (Suma de los primeros 4 bytes truncada a 8 bits)
149 uint8_t checksum = (uint8_t)(raw_bytes[0] + raw_bytes[1] + raw_bytes[2] + raw_bytes[3]);
150 if (checksum != raw_bytes[4]) {
151 out_data->lectura_valida = false;
152 return false; // Error de paridad o bit corrupto por ruido
153 }
154
155 // 10. Reconstrucción de la Humedad Relativa (16 bits sin signo, resolución de 0.1 %):
156 // Byte 0: Parte alta, Byte 1: Parte baja
157 out_data->hum_decimas_pct = (uint16_t)((raw_bytes[0] << 8) | raw_bytes[1]);
158
159 // 11. Reconstrucción de la Temperatura (16 bits con signo, resolución de 0.1 °C):
160 // Byte 2: Parte alta, Byte 3: Parte baja.
161 // El bit más significativo (bit 15) es el bit de signo (1 para temperaturas bajo cero).
162 uint16_t raw_temp = (uint16_t)((raw_bytes[2] << 8) | raw_bytes[3]);
163 if (raw_temp & 0x8000) {
164 // Temperatura negativa: Se limpia el bit de signo con máscara 0x7FFF y se aplica signo negativo
165 out_data->temp_decimas_c = -(int16_t)(raw_temp & 0x7FFF);
166 } else {
167 // Temperatura positiva
168 out_data->temp_decimas_c = (int16_t)raw_temp;
169 }
170
171 out_data->lectura_valida = true;
172 return true;
173}
